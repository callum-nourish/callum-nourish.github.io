```ruby
def sync_milestone_interactions(interaction_id, milestone_ids)
  MilestoneInteraction.transaction do
    # Get existing milestone associations for this interaction
    existing = MilestoneInteraction.where(interaction_id: interaction_id)
    
    # If no existing records, just create new ones
    if existing.empty?
      milestone_ids.each do |milestone_id|
        MilestoneInteraction.create!(
          interaction_id: interaction_id,
          milestone_id: milestone_id
        )
      end
      return
    end
    
    existing_milestone_ids = existing.pluck(:milestone_id)
    
    next if existing_milestone_ids.sort == milestone_ids.sort
    
    to_remove = existing_milestone_ids - milestone_ids
    existing.where(milestone_id: to_remove).delete_all if to_remove.any?
    
    to_add = milestone_ids - existing_milestone_ids
    to_add.each do |milestone_id|
      MilestoneInteraction.create!(
        interaction_id: interaction_id,
    existing_milestone_ids = existing.pluck(:milestone_id)

    to_remove = existing_milestone_ids - milestone_ids
    existing.where(milestone_id: to_remove).delete_all if to_remove.any?

      )
    end
  end
end
```
